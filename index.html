sheet.getRange(techStartRow, 1, 1, 2).setValues([['Technician', 'Inspections']]) .setFontWeight('bold').setBackground('#d4e3ef'); sheet.getRange(techStartRow + 1, 1, techRows.length, 2).setValues(techRows); sheet.insertChart(techChart); // ============================================================ // 5) WEEKLY TREND // ============================================================ const byWeek = {}; records.forEach(r => { const d = new Date(r['Timestamp']); if (weekRows.length > 0) { sheet.getRange(weekStartRow + 1, 1, weekRows.length, 4).setValues(weekRows); const weekChart = sheet.newChart() .setChartType(Charts.ChartType.LINE) .addRange(sheet.getRange(weekStartRow, 1, weekRows.length + 1, 4)) .setPosition(weekStartRow, 6, 0, 0) .setOption('title', 'Weekly Run Hours & Diesel Trend') .setOption('width', 600) .setOption('height', 380) .setOption('colors', ['#2a6b96', '#7a4c9c', '#1e4a6d']) .build(); sheet.insertChart(weekChart); } // ============================================================ // 6) FUEL DISTRIBUTION // ============================================================ const fuelBuckets = { 'Low (0–25%)': 0, 'Medium (26–50%)': 0, 'Good (51–75%)': 0, 'Full (76–100%)': 0 }; sheet.setColumnWidth(7, 60); sheet.setColumnWidth(8, 60); sheet.setColumnWidth(9, 60); sheet.setColumnWidth(10, 60); sheet.setFrozenRows(1); } // ============================================================ // 🛠️ HELPER FUNCTIONS // ============================================================ function parseNum(val) { if (val === null || val === undefined || val === '') return 0; if (typeof val === 'number') return val; const s = String(val).replace(/[^\d.-]/g, ''); const n = parseFloat(s); return isNaN(n) ? 0 : n; } function sumBy(arr, fn) { return arr.reduce((acc, item) => acc + (fn(item) || 0), 0); } function avgBy(arr, fn) { if (arr.length === 0) return 0; return sumBy(arr, fn) / arr.length; } function round(n, decimals) { const f = Math.pow(10, decimals); return Math.round(n * f) / f; } function groupBy(arr, keyFn) { const out = {}; arr.forEach(item => { const k = keyFn(item) || 'Unknown'; if (!out[k]) out[k] = []; out[k].push(item); }); return out; } function getISOWeekKey(date) { const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate())); const dayNum = d.getUTCDay() || 7; d.setUTCDate(d.getUTCDate() + 4 - dayNum); const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1)); const weekNo = Math.ceil((((d - yearStart) / 86400000) + 1) / 7); return d.getUTCFullYear() + '-W' + String(weekNo).padStart(2, '0'); }
